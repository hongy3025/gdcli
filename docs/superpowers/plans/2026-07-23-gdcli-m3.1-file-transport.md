# gdcli M3.1 File-Based Runtime Transport Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 完成 M3 runtime 验证闭环——在不依赖 Godot 4.7 headless 不支持的 `EngineDebugger` 远程调试通道的前提下，通过**编辑器/游戏进程共享文件系统上的原子文件交换**，让 `runtime/status` 在 headless harness 中到达 `connected`，并跑通所有 M3 E2E。

**Architecture:** 新增一对 fallback transport 模块，分别跑在游戏进程（probe 侧）和编辑器进程（plugin 侧）：
- `runtime_transport_file_probe.gd` 在 probe `_ready()` 阶段生成唯一 `probe_id`，写 `res://.godot/gdapi_runtime/<probe_id>/hello.json`，并在 `_process` 中轮询 `inbox/*.json`、把 reply 写到 `outbox/<id>.json`、按 `hello_delay_ms` 延迟发布。
- `runtime_transport_file_editor.gd` 在 plugin `_process` 中轮询 `res://.godot/gdapi_runtime/*/hello.json`；发现新 probe 时为 `GdApiRuntimeBroker` 注入 `attach_file_transport()` 返回的 send callable。

`GdApiRuntimeBroker` 已有 `attach(session_id, send)` 注入接口（commit 707e392 之后），协议 v1 schema 与 35 个 runtime 路由一行不动；只是当 EngineDebugger transport 在 headless 下不可达时，由文件 transport 接管同一套 broker 状态机与 callback 协议。

**Tech Stack:** Godot 4.7.x、`FileAccess` + `DirAccess`、`OS.get_unix_time_from_system`、`Time.get_ticks_msec`、`Engine.is_editor_hint()`、原子文件重命名。

## Global Constraints

- 仅支持 Godot 4.7.x；已完成 M1/M2 与本会话已落地的 M3 结构性修复（Step 0 / D1 / D5 / parse error 修复 / position 重命名）。
- `GdApiRuntimeBroker` 三态（stopped / connecting / connected）、pending 字典、timeout 处理、detach 幂等、id 单调递增——这些契约不动。
- protocol v1 schema（`runtime_protocol.gd::request/reply/event`）与 35 个 runtime 路由全部一行不动；probe 侧 `_dispatch_async` 仍按 op 字符串路由到 NodeOps / InputOps / CaptureOps。
- 单次 reply ≤ 4 MiB，单条消息体 ≤ 4 MiB，screenshot ≤ 1920x1080，sequence ≤ 100 events / 10 s——这些 4 MiB / 1920x1080 / 100 events / 10 s 上限与之前完全一致。
- 文件 transport 仅在 `EngineDebugger.is_active() == false` 时启用；编辑器自动选择活跃 transport，避免双通道重复发送。
- 文件 transport 路径必须使用 `ProjectSettings.globalize_path("res://.godot/gdapi_runtime/...")`，确保编辑器与游戏进程读写同一物理目录。
- 不修改 `tests/e2e/m3/conftest.py` 的 `m3_editor`/`m3_running` fixture 行为；现有 7 套 M3 E2E 测试除 timeout 放宽外断言一字不动。
- 测试 TDD：每个 task 先写失败测试，看到 RED，再写实现到 GREEN，最后 commit。

`gdapi/addon/runtime/runtime_transport_file_probe.gd`：

```gdscript
## 文件 transport —— 探针侧 (probe)
##
## 在游戏进程里通过共享文件系统与编辑器通信,绕开 Godot 4.7 headless
## 不支持的 EngineDebugger 远程调试通道。
##
## 目录布局（probe_id = 8 字符 hex）：
##   res://.godot/gdapi_runtime/<probe_id>/
##     hello.json       —— _ready 后立即写(或按 hello_delay_ms 推迟)
##     inbox/<id>.json  —— 编辑器写入的 request
##     outbox/<id>.json —— probe 处理后的 reply

@tool
class_name GdApiRuntimeTransportFileProbe
extends RefCounted

const Protocol := preload("res://addons/gdapi/runtime/runtime_protocol.gd")

## hello_delay_ms >0 时,start() 后推迟该毫秒数再写 hello.json
var hello_delay_ms: int = 0

## 自定义根目录(用于单元测试隔离);默认 res://.godot/gdapi_runtime
var root_dir_override: String = ""

## 唯一 probe_id,start() 时生成
var _probe_id: String = ""

## 是否已 start
var _started: bool = false

## request handler,由 editor 侧 bridge 调用;签名 (msg) -> reply
var _handler: Callable = Callable()

## hello 定时器到期时刻
var _hello_due_msec: int = 0
var _hello_sent: bool = false

func _init(p_hello_delay_ms: int = 0, p_root_override: String = "") -> void:
    hello_delay_ms = p_hello_delay_ms
    root_dir_override = p_root_override

## 暴露给测试 / 上层
func probe_id() -> String:
    if _probe_id.is_empty():
        _probe_id = _generate_probe_id()
    return _probe_id

## 返回根目录绝对路径
func root_path() -> String:
    if not root_dir_override.is_empty():
        return ProjectSettings.globalize_path(root_dir_override)
    return ProjectSettings.globalize_path("res://.godot/gdapi_runtime")

## 注册 request handler(msg) -> reply_dict
func set_request_handler(handler: Callable) -> void:
    _handler = handler

## 启动 transport:生成 probe_id,建子目录,写(或排程) hello.json
func start() -> void:
    if _started:
        return
    if _probe_id.is_empty():
        _probe_id = _generate_probe_id()
    var dir := root_path().path_join(_probe_id)
    DirAccess.make_dir_recursive_absolute(dir.path_join("inbox"))
    DirAccess.make_dir_recursive_absolute(dir.path_join("outbox"))
    _started = true
    if hello_delay_ms <= 0:
        _write_hello_file()
    else:
        _hello_due_msec = Time.get_ticks_msec() + hello_delay_ms

## 停止并清理
func stop() -> void:
    if not _started:
        return
    _started = false
    var dir := root_path().path_join(_probe_id)
    DirAccess.remove_absolute(dir)

## 每帧调用:处理 hello 推迟、inbox 扫描
func tick(now_msec: int) -> void:
    if not _started:
        return
    if not _hello_sent and hello_delay_ms > 0 and now_msec >= _hello_due_msec:
        _write_hello_file()
    _scan_inbox(now_msec)

## 私有:写 hello.json
func _write_hello_file() -> void:
    _hello_sent = true
    var hello := Protocol.event(0, "hello", {
        "protocol_version": Protocol.VERSION,
        "node": OS.get_processor_name(),
        "transport": "file",
    })
    var path := root_path().path_join(_probe_id).path_join("hello.json")
    _atomic_write(path, JSON.stringify(hello))

## 私有:扫描 inbox/ 处理每个 request
func _scan_inbox(_now_msec: int) -> void:
    var inbox := DirAccess.open(root_path().path_join(_probe_id).path_join("inbox"))
    if inbox == null:
        return
    for filename in inbox.get_files():
        if not filename.ends_with(".json"):
            continue
        var full_path := root_path().path_join(_probe_id).path_join("inbox").path_join(filename)
        var f := FileAccess.open(full_path, FileAccess.READ)
        if f == null:
            continue
        var raw := f.get_as_text()
        f.close()
        var dict: Dictionary = JSON.parse_string(raw) if not raw.is_empty() else {}
        if typeof(dict) != TYPE_DICTIONARY:
            inbox.remove(filename)
            continue
        var verdict := Protocol.validate_message(dict)
        if not bool(verdict.get("ok", false)):
            inbox.remove(filename)
            continue
        if String(dict.get("kind", "")) != "request":
            inbox.remove(filename)
            continue
        # 删除 inbox 防止重复消费
        inbox.remove(filename)
        var reply: Dictionary = _dispatch_request(dict)
        var reply_msg := Protocol.reply(int(dict.id), bool(reply.get("ok", false)),
            reply.get("result", {}), String(reply.get("error", "")),
            String(reply.get("code", "")))
        var out_path := root_path().path_join(_probe_id).path_join("outbox").path_join(str(int(dict.id)) + ".json")
        _atomic_write(out_path, JSON.stringify(reply_msg))

## 私有:分派 request 到已注册 handler,或返回 not_supported
func _dispatch_request(req: Dictionary) -> Dictionary:
    if not _handler.is_valid():
        return {"ok": false, "code": "not_supported", "error": "no handler registered"}
    return _handler.call(req)

## 私有:原子写（先写 .tmp 再 rename）
func _atomic_write(target_path: String, content: String) -> void:
    var tmp_path := target_path + ".tmp"
    var f := FileAccess.open(tmp_path, FileAccess.WRITE)
    if f == null:
        return
    f.store_string(content)
    f.close()
    DirAccess.rename_absolute(tmp_path, target_path)

## 私有:生成 8 字符 hex probe id
func _generate_probe_id() -> String:
    var hex := "0123456789abcdef"
    var out := ""
    var rng := RandomNumberGenerator.new()
    rng.randomize()
    for i in 8:
        out += hex[rng.randi() % 16]
    return out
```

```gdscript
## 文件 transport —— 编辑器侧 (plugin)
##
## 每帧扫描 res://.godot/gdapi_runtime/*/hello.json 发现 probe;
## 通过 send callable 注入 GdApiRuntimeBroker,与 EngineDebugger transport
## 并存(优先 EngineDebugger;headless 下回落到 file)。

@tool
class_name GdApiRuntimeTransportFileEditor
extends RefCounted

const Protocol := preload("res://addons/gdapi/runtime/runtime_protocol.gd")

var root_dir_override: String = ""
var _broker: RefCounted = null
var _probes: Dictionary = {}  ## probe_id -> {send: Callable, attached: bool}
var _pending: Dictionary = {} ## id -> {probe_id, deadline_msec, callback, op}
var _next_id: int = 0
var _started: bool = false

func _init(p_root_override: String = "") -> void:
    root_dir_override = p_root_override

func setup(broker: RefCounted) -> void:
    _broker = broker

func root_path() -> String:
    if not root_dir_override.is_empty():
        return ProjectSettings.globalize_path(root_dir_override)
    return ProjectSettings.globalize_path("res://.godot/gdapi_runtime")

func start() -> void:
    _started = true
    DirAccess.make_dir_recursive_absolute(root_path())

func stop_all(reason: String = "editor transport stopping") -> void:
    _started = false
    for id in _pending.keys():
        var entry: Dictionary = _pending[id]
        var cb: Callable = entry.callback
        if cb.is_valid():
            cb.call({"ok": false, "code": "conflict", "error": reason, "request_id": int(id)})
    _pending.clear()
    _probes.clear()

func active_probe_ids() -> Array:
    return _probes.keys()

func pending_count() -> int:
    return _pending.size()

func request(op: String, payload: Dictionary, timeout_ms: int, on_complete: Callable) -> int:
    if _probes.is_empty():
        on_complete.call({"ok": false, "code": "conflict",
            "error": "no runtime probe connected", "request_id": -1})
        return -1
    var probe_id: String = _probes.keys()[0]  # 单 probe 设计
    _next_id += 1
    var id := _next_id
    var msg := Protocol.request(id, op, payload)
    var deadline := Time.get_ticks_msec() + (timeout_ms if timeout_ms > 0 else 5000)
    _pending[id] = {
        "probe_id": probe_id,
        "deadline_msec": deadline,
        "callback": on_complete,
        "op": op,
    }
    var inbox_path := root_path().path_join(probe_id).path_join("inbox").path_join(str(id) + ".json")
    _atomic_write(inbox_path, JSON.stringify(msg))
    return id

func tick(now_msec: int) -> void:
    if not _started:
        return
    _scan_hello_files()
    _scan_outbox()
    _expire_timeouts(now_msec)

func _scan_hello_files() -> void:
    var dir := DirAccess.open(root_path())
    if dir == null:
        return
    for name in dir.get_directories():
        var hello_path := root_path().path_join(name).path_join("hello.json")
        if not FileAccess.file_exists(hello_path):
            continue
        if _probes.has(name):
            continue
        _attach_probe(name)

func _attach_probe(probe_id: String) -> void:
    if _broker == null:
        _probes[probe_id] = {"attached": false}
        return
    var self_ref := self
    var send := func(msg: Dictionary) -> bool:
        var inbox_path := root_path().path_join(probe_id).path_join("inbox").path_join(str(int(msg.id)) + ".json")
        self_ref._atomic_write(inbox_path, JSON.stringify(msg))
        return true
    if _broker.has_method("attach_file_transport"):
        _broker.call("attach_file_transport", probe_id, send)
    _probes[probe_id] = {"attached": true}

func _scan_outbox() -> void:
    for probe_id in _probes.keys():
        var out_dir := DirAccess.open(root_path().path_join(probe_id).path_join("outbox"))
        if out_dir == null:
            continue
        for filename in out_dir.get_files():
            if not filename.ends_with(".json"):
                continue
            var id_str := filename.get_basename()
            if not id_str.is_valid_int():
                out_dir.remove(filename)
                continue
            var id := int(id_str)
            if not _pending.has(id):
                out_dir.remove(filename)
                continue
            var full := root_path().path_join(probe_id).path_join("outbox").path_join(filename)
            var f := FileAccess.open(full, FileAccess.READ)
            if f == null:
                continue
            var raw := f.get_as_text()
            f.close()
            var dict: Dictionary = JSON.parse_string(raw) if not raw.is_empty() else {}
            out_dir.remove(filename)
            if typeof(dict) != TYPE_DICTIONARY:
                continue
            var entry: Dictionary = _pending[id]
            _pending.erase(id)
            var cb: Callable = entry.callback
            if cb.is_valid():
                cb.call(dict)

func _expire_timeouts(now_msec: int) -> void:
    var expired: Array = []
    for id in _pending.keys():
        if int(_pending[id].deadline_msec) <= now_msec:
            expired.append(id)
    for id in expired:
        var entry: Dictionary = _pending[id]
        _pending.erase(id)
        var cb: Callable = entry.callback
        if cb.is_valid():
            cb.call({"ok": false, "code": "timeout",
                "error": "runtime request timed out", "request_id": int(id)})

func _atomic_write(target_path: String, content: String) -> void:
    var tmp_path := target_path + ".tmp"
    var f := FileAccess.open(tmp_path, FileAccess.WRITE)
    if f == null:
        return
    f.store_string(content)
    f.close()
    DirAccess.rename_absolute(tmp_path, target_path)
```

## 风险与回退

- **文件 transport 与 EngineDebugger transport 并存**:headless 下 EngineDebugger 仍走 `register_message_capture` 但永远收不到 game_view 以外的回包,file transport 兜底。非 headless 下 EngineDebugger 先到,file transport 一直轮询空目录(无副作用)。`detach` 任一 transport 都会触发 broker.detach,不会双通道并发 reply。
- **probe_id 唯一性**:8 字符 hex = 16^8 ≈ 4.3 × 10^9 组合;同时运行的探针数量极少,碰撞概率忽略;若真碰撞,两个 probe 会互相消费对方的 outbox 文件,造成 request id 错位——可在 unit test 加碰撞检测,如果观察到碰撞再加 probe_id = pid + 时间戳。
- **probe `_process` 轮询开销**:每个 game tick 扫一次 inbox 目录。空目录扫一次成本 < 1ms;inbox 有文件时逐个处理,单条 < 5ms。可接受。
- **m3_running fixture 的 lifecycle 超时**:原 15s 在 file transport 下可能不够(file transport roundtrip < 100ms,但 conftest 在 setup 期 + game 启动 + probe ready 累计可能 ≥ 5s);已统一放宽到 30s。
- **autoload 竞速 (D6)**:在 file transport 下完全不相关——file transport 不依赖 EngineDebugger session 打开顺序。
- **回退**:若 file transport 在生产环境表现不佳(例如 NFR 要求毫秒级 roundtrip),可重新评估 TCP socket transport;但当前 M3 5s timeout + file transport < 100ms 已留出 50x 余量,不会成为瓶颈。
